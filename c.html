Q5 Demonstrate loop

# Program to demonstrate different loops in Python

# FOR LOOP
print("For Loop Example:")
for i in range(1, 6):
    print("Number:", i)

# WHILE LOOP
print("\nWhile Loop Example:")
j = 1
while j <= 5:
    print("Number:", j)
    j += 1

# NESTED LOOP
print("\nNested Loop Example:")
for i in range(1, 4):
    for j in range(1, 4):
        print(i, j)

# LOOP WITH CONDITION
print("\nEven Numbers from 1 to 10:")
for i in range(1, 11):
    if i % 2 == 0:
        print(i)

# method overriding 

class base:
    def show(self):
        print("this is base class ")

class derived(base):
    def show(self):
        print("this is child class ")

d=derived()
d.show()
base.show(d) #calling method for base class
