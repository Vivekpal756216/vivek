STRING FUNCTION 

                
                
            
<!DOCTYPE html>
<html>
<head>
<title>String Functions</title>
</head>

<body>

<script>
// var s = "ABC PQ ABcd";
var s = new String("ABC PQ ABcd");
var s1 = "abc", s2 = "pqrs";

var len = s.length;
document.write("Length=" + len + "<br>");

document.write("2nd char :" + s.charAt(2) + "<br>");
document.write("AB at :" + s.indexOf("AB") + "<br>");
document.write("AB after 2 :" + s.indexOf("AB", 2) + "<br>");
document.write("last AB at :" + s.lastIndexOf("AB") + "<br>");
document.write("Sub-string = " + s.substring(2, 8) + "<br>");

document.write("concat :" + s1.concat(s2) + "<br>");
document.write("concat 2 :" + s1.concat(s2, s) + "<br>");

document.write("Slice :" + s.slice(2) + "<br>");

var a = s.split("B");
var i;

document.write("Parts .. <br>");
for (i = 0; i < a.length; i++)
    document.write(a[i] + "<br>");

document.write("upper case = " + s.toUpperCase() + "<br>");
document.write("repeat :" + s1.repeat(3) + "<br>");
</script>

</body>
</html>
